CREATE TABLE TUTOR (
	id_tutor integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	nome varchar(100), 
	telefone char(15),
	email varchar(50),
	endereco varchar(254)
)

SELECT * FROM TUTOR 

CREATE TABLE ANIMAL (
    ID_ANIMAL INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR(50),
    RACA VARCHAR(50),
    IDADE INTEGER,
    ESPECIE VARCHAR(50),
    ID_TUTOR INTEGER
)

SELECT * FROM ANIMAL 

CREATE TABLE VETERINARIO (
    ID_VETERINARIO INTEGER GENERATED BY DEFAULT AS IDENTITY  PRIMARY KEY,
    NOME VARCHAR(50),
    ESPECIALIDADE VARCHAR(50),
    TELEFONE INTEGER
)

SELECT * FROM VETERINARIO  

CREATE TABLE AGENDAMENTO (
    ID_AGENDAMENTO INTEGER GENERATED BY DEFAULT AS IDENTITY  PRIMARY KEY,
    MOTIVO_CONSULTA VARCHAR(50),
    DATA_E_HORA TIMESTAMP,
    ID_TUTOR INTEGER,
    ID_ANIMAL INTEGER,
    ID_VETERINARIO INTEGER
)

SELECT * FROM AGENDAMENTO  

CREATE TABLE RECEITA (
    ID_RECEITA INTEGER GENERATED BY DEFAULT AS IDENTITY  PRIMARY KEY,
    DATA_EMISSAO DATE,
    OBSERVACOES VARCHAR(100),
    ID_AGENDAMENTO INTEGER
)

SELECT * FROM RECEITA  

CREATE TABLE MEDICAMENTO (
    ID_MEDICAMENTO INTEGER GENERATED BY DEFAULT AS IDENTITY  PRIMARY KEY,
    NOME VARCHAR(50),
    FABRICANTE VARCHAR(50),
    TIPO VARCHAR(50)
)

SELECT * FROM  MEDICAMENTO
